set(NETEQ_ROOT "${PROJECT_SOURCE_DIR}/research/upstream/webrtc-neteq")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/prepare_neteq.py"
    COMMAND_ERROR_IS_FATAL ANY)
include("${PROJECT_SOURCE_DIR}/build/neteq-sources.cmake")
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${PROJECT_SOURCE_DIR}/research/upstream/abseil-cpp" "${PROJECT_BINARY_DIR}/abseil" EXCLUDE_FROM_ALL)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|aarch64)")
    list(FILTER ALISA_NETEQ_SOURCES EXCLUDE REGEX "_(sse|avx)")
else()
    foreach(source IN LISTS ALISA_NETEQ_SOURCES)
        if(source MATCHES "_avx2\\.cc$")
            set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
        endif()
    endforeach()
endif()
add_library(alisa-neteq STATIC ${ALISA_NETEQ_SOURCES} "${PROJECT_SOURCE_DIR}/native/neteq_c.cpp")
target_include_directories(alisa-neteq PRIVATE "${NETEQ_ROOT}" "${PROJECT_SOURCE_DIR}/native" "${PROJECT_SOURCE_DIR}/build/arm/prefix/include/opus")
target_compile_definitions(alisa-neteq PRIVATE WEBRTC_POSIX WEBRTC_LINUX WEBRTC_ABSL_MUTEX WEBRTC_ENABLE_PROTOBUF=0
    WEBRTC_STRICT_FIELD_TRIALS=0 WEBRTC_CODEC_OPUS WEBRTC_OPUS_SUPPORT_120MS_PTIME=1 WEBRTC_OPUS_VARIABLE_COMPLEXITY=0 RTC_DISABLE_TRACE_EVENTS)
target_compile_options(alisa-neteq PRIVATE -O2 -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections)
if(CMAKE_CROSSCOMPILING)
    set(NETEQ_OPUS "${PROJECT_SOURCE_DIR}/build/arm/prefix/lib/libopus.a")
else()
    find_library(NETEQ_OPUS opus REQUIRED)
endif()
target_link_libraries(alisa-neteq PUBLIC ${ALISA_NETEQ_ABSL} "${NETEQ_OPUS}" pthread m)
add_executable(alisa-neteq-check "${PROJECT_SOURCE_DIR}/tests/neteq.cpp")
target_include_directories(alisa-neteq-check PRIVATE "${PROJECT_SOURCE_DIR}/native"
    "${PROJECT_SOURCE_DIR}/build/arm/prefix/include/opus"
    "${PROJECT_SOURCE_DIR}/research/upstream/libdatachannel/deps/json/single_include")
target_compile_options(alisa-neteq-check PRIVATE -UNDEBUG)
target_link_libraries(alisa-neteq-check PRIVATE alisa-neteq)
add_executable(alisa-neteq-device-check "${PROJECT_SOURCE_DIR}/tests/neteq_device.cpp")
target_include_directories(alisa-neteq-device-check PRIVATE "${PROJECT_SOURCE_DIR}/native"
    "${PROJECT_SOURCE_DIR}/build/arm/prefix/include/opus"
    "${PROJECT_SOURCE_DIR}/research/upstream/libdatachannel/deps/json/single_include")
target_link_libraries(alisa-neteq-device-check PRIVATE alisa-neteq dl)
